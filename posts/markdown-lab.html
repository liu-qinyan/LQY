1. 代码块与高亮 ​

给看板娘喂冻干的伪代码（TypeScript 味儿）：

ts
interface CatGirl {
  name: string
  mood: 'sleepy' | 'playful' | 'tsundere'
}

function feed(cat: CatGirl, snack: string): string {
  if (cat.mood === 'sleepy') return `${cat.name} 翻了个身，假装没听见`
  return `${cat.name} 两眼放光地扑向了 ${snack}！喵！`
}

const shizuku: CatGirl = { name: '雫雫', mood: 'tsundere' }
console.log(feed(shizuku, '猫条'))
// "雫雫 两眼放光地扑向了 猫条！喵！"（嘴上傲娇，身体很诚实）
